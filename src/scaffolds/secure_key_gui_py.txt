Modern, flat (MUI-style) GUI front-end for secure_key.py
Tkinter + ttk only, zero extra dependencies.
"""

import tkinter as tk
from tkinter import ttk
import keyring
from keyring.errors import PasswordDeleteError

from secure_key import _provider_service, _encrypt, _cache_passphrase
from constants   import TOKEN_KEYS          # list[str] | list[tuple[str,str]]

PRIMARY      = "#1976D2"
PRIMARY_HOV  = "#1565C0"
PRIMARY_PRS  = "#0D47A1"
BG_WINDOW    = "#F4F6F8"
FG_LABEL     = "#5F6368"
FG_TEXT      = "#202124"
FONT_UI      = ("Segoe UI", 10)

class SetupGUI(tk.Tk):
    def __init__(self) -> None:
        super().__init__()
        # — window
        self.title(TITLE)
        self.iconphoto(True, tk.PhotoImage(data=ICON_B64))
        self.configure(background=BG_WINDOW)
        self.resizable(False, False)

        # — global style ---------------------------------------------------
        style = ttk.Style(self)
        style.theme_use("clam")
        style.configure(".", font=FONT_UI, background=BG_WINDOW)
        style.configure("TLabel", foreground=FG_LABEL, background=BG_WINDOW)
        style.configure("Flat.TEntry",
                        relief="flat", padding=4,
                        fieldbackground="white", foreground=FG_TEXT)
        style.configure("Primary.TButton",
                        foreground="white", background=PRIMARY,
                        padding=(14, 6), borderwidth=0)
        style.map("Primary.TButton",
                  background=[("active", PRIMARY_HOV),
                              ("pressed", PRIMARY_PRS)])

        # — card frame ------------------------------------------------------
        card = ttk.Frame(self, padding=24)
        card.grid()
        card.columnconfigure(1, weight=1)

        self.inputs: dict[str, ttk.Entry] = {}

        # ---------- token fields ------------------------------------------
        for row, item in enumerate(TOKEN_KEYS):
            # accept either "OPENAI_API_KEY" or ("OPENAI_API_KEY", "OpenAI key")
            if isinstance(item, tuple):
                key, label = item
            else:
                key   = item
                label = key.replace("_API_KEY", "").replace("_", " ").title() + " API key"

            ttk.Label(card, text=label + ":").grid(
                row=row, column=0, sticky="e", padx=(0, 12), pady=4
            )
            ent = ttk.Entry(card, width=38,
                            show="*" if "KEY" in key else "",
                            style="Flat.TEntry")
            ent.grid(row=row, column=1, pady=4, sticky="ew")
            self.inputs[key] = ent

        # ---------- pass-phrase -------------------------------------------
        last = len(TOKEN_KEYS)
        ttk.Label(card, text="Pass-phrase:").grid(
            row=last, column=0, sticky="e", padx=(0, 12), pady=(12, 4)
        )
        self.pass_entry = ttk.Entry(card, width=38, show="*", style="Flat.TEntry")
        self.pass_entry.grid(row=last, column=1, pady=(12, 4), sticky="ew")

        # ---------- action buttons ----------------------------------------
        btn_row = last + 1
        save_btn = ttk.Button(card, text="Save", style="Primary.TButton",
                              command=self.save)
        save_btn.grid(row=btn_row, column=0, sticky="ew", pady=(16, 4))

        clear_btn = ttk.Button(card, text="Clear cache",
                               command=self.clear_cache)
        clear_btn.grid(row=btn_row, column=1, sticky="ew", pady=(16, 4))

        # ---------- inline status banner ----------------------------------
        self.status_lbl = ttk.Label(card, text="", anchor="center")
        self.status_lbl.grid(row=btn_row + 1, column=0, columnspan=2,
                             sticky="ew", pady=(4, 2))

        self.update_idletasks()
        self.minsize(self.winfo_width(), self.winfo_height())

    # ──────────────────  save secrets  ────────────────────────────
    def save(self) -> None:
        phrase = self.pass_entry.get().strip()
        if not phrase:
            self._status("Pass-phrase cannot be empty.", "red")
            return

        saved = 0
        for key, entry in self.inputs.items():
            val = entry.get().strip()
            if not val:
                continue
            ns = _provider_service(key)
            keyring.set_password(ns, key, _encrypt(val, phrase))
            _cache_passphrase(ns, phrase)
            saved += 1

        if saved:
            self._status(f"✔ Stored {saved} secret(s).", "green")
        else:
            self._status("No values entered.", "orange")

    # ──────────────────  clear cache  ────────────────────────────
    def clear_cache(self) -> None:
        deleted = 0
        for item in TOKEN_KEYS:
            key = item[0] if isinstance(item, tuple) else item
            ns  = _provider_service(key)
            try:
                keyring.delete_password(ns, key)
                deleted += 1
            except PasswordDeleteError:
                pass    # not present – ignore
            _cache_passphrase(ns, None)   # clear cached passphrase

        if deleted:
            self._status(f"🗑  Cleared {deleted} key(s).", "green")
            for ent in self.inputs.values():
                ent.delete(0, tk.END)
        else:
            self._status("Nothing to clear.", "orange")

    # ──────────────────  helper  ─────────────────────────────────
    def _status(self, msg: str, color: str) -> None:
        self.status_lbl.configure(text=msg, foreground=color)

# ────────────────────────────────────────────────────────────────
if __name__ == "__main__":
    SetupGUI().mainloop()
