# ─────────────────────────── GUI class ───────────────────────────────
class SetupGUI(tk.Tk):
    def __init__(self) -> None:
        super().__init__()
        # window basics
        self.title(TITLE)
        self.iconphoto(True, tk.PhotoImage(data=ICON_B64))
        self.configure(background=BG_WINDOW)
        self.resizable(False, False)

        # global ttk style
        style = ttk.Style(self)
        style.theme_use("clam")
        style.configure(".", font=FONT_UI, background=BG_WINDOW)
        style.configure("TLabel", foreground=FG_LABEL, background=BG_WINDOW)
        style.configure(
            "Flat.TEntry",
            relief="flat",
            padding=4,
            fieldbackground="white",
            foreground=FG_TEXT,
        )
        style.configure(
            "Primary.TButton",
            foreground="white",
            background=PRIMARY,
            padding=(14, 6),
            borderwidth=0,
        )
        style.map(
            "Primary.TButton",
            background=[("active", PRIMARY_HOV), ("pressed", PRIMARY_PRS)],
        )

        # card frame
        card = ttk.Frame(self, padding=24)
        card.grid()
        card.columnconfigure(1, weight=1)

        self.inputs: dict[str, ttk.Entry] = {}

        # ---------- token fields --------------------------------------
        for row, item in enumerate(TOKEN_KEYS):
            key, label = (
                item if isinstance(item, tuple) else (item, item.replace("_API_KEY", "").replace("_", " ").title() + " API key")

            ttk.Label(card, text=label + ":").grid(
                row=row, column=0, sticky="e", padx=(0, 12), pady=4
            )
            ent = ttk.Entry(
                card,
                width=38,
                show="*" if "KEY" in key else "",
                style="Flat.TEntry",
            )
            ent.grid(row=row, column=1, pady=4, sticky="ew")
            self.inputs[key] = ent

        cur_row = len(TOKEN_KEYS)

        # ---------- engine selector -----------------------------------
        ttk.Label(card, text="Crypto engine:").grid(
            row=cur_row, column=0, sticky="e", padx=(0, 12), pady=(12, 4)
        )
        self.engine_box = ttk.Combobox(
            card,
            values=list(ENGINES.keys()),
            state="readonly",
            width=35,
        )
        self.engine_box.set(DEFAULT_ENGINE)
        self.engine_box.grid(row=cur_row, column=1, pady=(12, 4), sticky="ew")
        cur_row += 1

        # ---------- TTL ----------------------------------------------
        ttk.Label(card, text="Cache TTL (sec):").grid(
            row=cur_row, column=0, sticky="e", padx=(0, 12), pady=4
        )
        self.ttl_entry = ttk.Entry(card, width=38, style="Flat.TEntry")
        self.ttl_entry.insert(0, "900")
        self.ttl_entry.grid(row=cur_row, column=1, pady=4, sticky="ew")
        cur_row += 1

        # ---------- pass-phrase --------------------------------------
        ttk.Label(card, text="Pass-phrase:").grid(
            row=cur_row, column=0, sticky="e", padx=(0, 12), pady=(12, 4)
        )
        self.pass_entry = ttk.Entry(card, width=38, show="*", style="Flat.TEntry")
        self.pass_entry.grid(row=cur_row, column=1, pady=(12, 4), sticky="ew")
        cur_row += 1

        # ---------- action buttons -----------------------------------
        save_btn = ttk.Button(
            card, text="Save", style="Primary.TButton", command=self.save
        )
        save_btn.grid(row=cur_row, column=0, sticky="ew", pady=(16, 4))

        clear_btn = ttk.Button(card, text="Clear cache", command=self.clear_cache)
        clear_btn.grid(row=cur_row, column=1, sticky="ew", pady=(16, 4))
        cur_row += 1

        # ---------- status banner ------------------------------------
        self.status_lbl = ttk.Label(card, text="", anchor="center")
        self.status_lbl.grid(row=cur_row, column=0, columnspan=2, sticky="ew", pady=(4, 2))

        self.update_idletasks()
        self.minsize(self.winfo_width(), self.winfo_height())

    # ───────────────────── save secrets ────────────────────────────
    def save(self) -> None:
        phrase = self.pass_entry.get().strip()
        if not phrase:
            self._status("Pass-phrase cannot be empty.", "red")
            return

        try:
            ttl = int(self.ttl_entry.get())
            if ttl < 0:
                raise ValueError
        except ValueError:
            self._status("TTL must be a non-negative integer.", "red")
            return

        engine = self.engine_box.get()
        if engine not in ENGINES:
            self._status("Invalid engine selected.", "red")
            return

        saved = 0
        for key, entry in self.inputs.items():
            val = entry.get().strip()
            if not val:
                continue
            ns = _provider_service(key)
            keyring.set_password(ns, key, _encrypt(val, phrase, engine))
            _cache_passphrase(ns, phrase, ttl)
            saved += 1

        if saved:
            self._status(f"✔ Stored {saved} secret(s).", "green")
        else:
            self._status("No values entered.", "orange")

    # ───────────────────── clear cache ─────────────────────────────
    def clear_cache(self) -> None:
        deleted = 0
        for item in TOKEN_KEYS:
            key = item[0] if isinstance(item, tuple) else item
            ns = _provider_service(key)
            # delete ciphertext
            try:
                keyring.delete_password(ns, key)
                deleted += 1
            except PasswordDeleteError:
                pass
            # delete pass-phrase cache
            if os.name == "nt":
                _cache_path(ns).unlink(missing_ok=True)
            else:
                try:
                    keyring.delete_password(_TOKEN_SVC, f"{ns}/_shared")
                except PasswordDeleteError:
                    pass
        self._status(
            f"{'🗑  Cleared' if deleted else 'Nothing to clear.'}", "green" if deleted else "orange"
        )
        for ent in self.inputs.values():
            ent.delete(0, tk.END)

    # ───────────────────── helper ────────────────────────────────
    def _status(self, msg: str, color: str) -> None:
        self.status_lbl.configure(text=msg, foreground=color)


# ─────────────────────────── run as script ───────────────────────────
if __name__ == "__main__":
    SetupGUI().mainloop()
